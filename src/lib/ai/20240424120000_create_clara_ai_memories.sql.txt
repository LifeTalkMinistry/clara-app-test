-- Migration: Create CLARA AI memories table
--
-- This migration adds a new table to store extracted user memories for
-- personalised financial coaching. Each record represents a single
-- piece of context that the user has shared with the assistant.

create table if not exists public.clara_ai_memories (
  id uuid primary key default uuid_generate_v4(),
  user_id uuid references auth.users(id) on delete cascade,
  category text not null,
  content text not null,
  inserted_at timestamp with time zone default now()
);

create index if not exists idx_clara_ai_memories_user on public.clara_ai_memories (user_id);
